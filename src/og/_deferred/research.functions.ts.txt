import { createServerFn } from "@og/compat/server-fn";
import { z } from "zod";
import { requireSupabaseAuth } from "@og/integrations/supabase/auth-middleware";
import { webSearch } from "./research.server";

export const runWebSearch = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input) =>
    z.object({ query: z.string().min(1), limit: z.number().int().min(1).max(20).default(6) }).parse(input),
  )
  .handler(async ({ data }) => {
    try {
      const results = await webSearch(data.query, data.limit, false);
      return { ok: true as const, sources: results.map((r) => ({ url: r.url, title: r.title ?? r.url, description: r.description })) };
    } catch (e) {
      return { ok: false as const, error: e instanceof Error ? e.message : String(e) };
    }
  });
