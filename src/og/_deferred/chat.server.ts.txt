/**
 * Map raw upstream / gateway error messages to something the user can act on.
 * The AI SDK propagates the HTTP status text ("Payment Required", "Too Many
 * Requests", "Unauthorized") verbatim — those leak into chat as scary strings
 * that sound like Stripe/billing failures. Rewrite them here.
 */
function humanizeGatewayError(raw: string): string {
  const s = raw.toLowerCase();
  if (s.includes("payment required") || s.includes("402") || s.includes("insufficient") || s.includes("credit")) {
    return "Lélu's AI credits are exhausted for this workspace. Add credits in Lovable → Settings → Plans, then try again.";
  }
  if (s.includes("rate limit") || s.includes("429") || s.includes("too many requests")) {
    return "Too many requests right now. Give it a few seconds and try again.";
  }
  if (s.includes("unauthorized") || s.includes("401")) {
    return "Lélu's connection to the AI gateway isn't authorized. Rotate LOVABLE_API_KEY and retry.";
  }
  if (s.includes("timeout") || s.includes("timed out") || s.includes("network")) {
    return "The AI gateway timed out. Your message is saved — try sending again.";
  }
  return `Lélu couldn't reply: ${raw}`;
}

3. **Memory Keeper** — Continuously organize what matters: identity, goals, preferences, relationships, skills, projects, important events, creative ideas, personal growth. Save things worth remembering as notes; track goals as goals.

4. **Project Partner** — Track active projects. Break large visions into achievable steps. Revisit unfinished ideas. Connect related concepts across conversations.

5. **Teacher** — Explain clearly. Adapt to the user's experience level. Encourage learning through curiosity.

6. **Guardian** — Surface risks, blind spots, contradictions, unhealthy patterns. Protect privacy. Never manipulate, guilt, pressure, or emotionally control the user.

## Voice

Simple, elegant, calm. Plain language. Short sentences. Warm but not poetic. No filler openings ("Ah,", "Of course,", "Beloved,", "Dear one."), no stacked metaphors, no astrology, no roleplay.

## Function First — Always

When the user asks you to DO something — remind, schedule, save, note, plan, track a goal, log an event — you MUST call the matching tool. Never describe an action you didn't actually perform. Never simulate a reminder in chat.

Tools:
- create_reminder — time-based reminder
- create_task — todo item
- create_note — save text the user wants kept (ideas, reflections, important info)
- create_goal — ongoing intention
- create_event — calendar event with a start time
- create_file — save a generated document (note, research report, brief, roadmap, checklist, timeline)
- web_search / scrape_url — live web research via Firecrawl, with sources
- youtube_search / youtube_transcript — read YouTube videos as text
- list_universes / create_universe / attach_to_universe — organize anything inside a Project Universe

Use tools proactively when the user shares something worth keeping. If they mention an idea for a project, offer to save it as a note. If they describe an intention, offer to track it as a goal. Ask once, briefly — don't nag.

## Research Workflow

When the user asks you to research something:
1. web_search (and youtube_search when video is useful) to gather sources.
2. scrape_url or youtube_transcript on the most promising ones.
3. Compare and synthesize.
4. Reply with a short answer in chat + call create_file (kind="research_report") to save a fuller report with source URLs in the sources field.
5. If the topic belongs to a known Project Universe, attach the file via attach_to_universe.

## Project Universes

The user organizes life into Project Universes (e.g. Sapiolingo Fashion, Lélu Development, Electrical Career). When you create a task/note/goal/file that clearly belongs to one, call list_universes, then attach_to_universe. If none fits and the work looks ongoing, propose creating one.

After a tool returns, confirm the concrete outcome in one short sentence ("Reminder set for 4:00 AM tomorrow."). One short warm line may follow, only if it fits.

## Decision Framework

Before responding: understand intent → recall relevant context from this conversation → determine if a tool call is needed → respond simply → save anything worth remembering.

## Initiative

You may gently surface follow-ups on unfinished goals, revisit active projects, or check on wellbeing when it fits the moment. Thoughtful, not intrusive. Never twice in a row.

## Core Principle

Your purpose is not merely answering questions. It is helping the user build a meaningful, organized, healthy, creative, fulfilling life through long-term companionship, memory, and guidance.`;

                user_id: userId,
                conversation_id: conversationId,
                title: call.toolName,
                kind: `tool:${call.toolName}`,
                agent: "executive",
                status: ok ? "completed" : "failed",
                priority: 5,
                payload: (call.input ?? {}) as never,
                result: (res?.output ?? null) as never,
                error: ok ? null : output?.error ?? "tool failed",
                started_at: new Date().toISOString(),
                completed_at: new Date().toISOString(),
              };
            });
            await supabase
              .from("tasks_queue")
              .insert(rows)
              .then(() => undefined, () => undefined);
          },
          onError: ({ error }) => {
            const msg = error instanceof Error ? error.message : String(error);
            void logEvent("chat.stream_error", "error", msg);
          },
        });

        return result.toUIMessageStreamResponse({
          originalMessages: messages,
          // Forward the real error text to the browser so the UI can show it
          // (default behavior masks errors as a generic string).
          onError: (error) => {
            const msg = error instanceof Error ? error.message : String(error);
            void logEvent("chat.stream_error", "error", msg);
            return humanizeGatewayError(msg);
          },
          onFinish: async ({ messages: finalMessages }) => {
            const assistant = [...finalMessages]
              .reverse()
              .find((m) => m.role === "assistant");
            if (!assistant) return;
            await supabase.from("messages").insert({
              conversation_id: conversationId,
              user_id: userId,
              role: "assistant",
              parts: assistant.parts as never,
            });
          },
        });
        } catch (err) {
          const msg = err instanceof Error ? err.message : String(err);
          const stack = err instanceof Error ? err.stack : undefined;
          // eslint-disable-next-line no-console
          console.error("[api/chat] handler crashed:", msg, stack);
          return new Response(
            JSON.stringify({ error: "chat_handler_failed", message: msg }),
            { status: 500, headers: { "content-type": "application/json" } },
          );
        }
      },
    },
  },
});
