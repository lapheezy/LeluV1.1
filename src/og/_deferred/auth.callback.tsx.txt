import { createFileRoute } from "@og/compat/router";
import { useEffect, useState } from "react";
import { supabase } from "@og/integrations/supabase/client";

export const Route = createFileRoute("/auth/callback")({
  ssr: false,
  head: () => ({
    meta: [
      { title: "Signing you in — Lélu" },
      { name: "description", content: "Finishing your sign-in and opening Lélu's core." },
      { property: "og:title", content: "Signing you in — Lélu" },
      { property: "og:description", content: "Finishing your sign-in and opening Lélu's core." },
      { property: "og:type", content: "website" },
      { name: "twitter:card", content: "summary" },
    ],
  }),
  component: AuthCallback,
});

function AuthCallback() {
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let active = true;
    const timeout = window.setTimeout(() => {
      if (!active) return;
      setError("Google sign-in did not finish. Please return to the horizon and try again.");
    }, 10000);

    const finish = async () => {
      const { data, error: sessionError } = await supabase.auth.getSession();
      if (!active) return;
      if (sessionError) {
        window.clearTimeout(timeout);
        setError(sessionError.message);
        return;
      }
      if (!data.session) return;
      window.clearTimeout(timeout);
      sessionStorage.removeItem("lelu:pending-auth");
      window.location.replace("/core");
    };

    const { data: listener } = supabase.auth.onAuthStateChange((_event, session) => {
      if (!active || !session) return;
      window.clearTimeout(timeout);
      sessionStorage.removeItem("lelu:pending-auth");
      window.location.replace("/core");
    });
    void finish();

    return () => {
      active = false;
      window.clearTimeout(timeout);
      listener.subscription.unsubscribe();
    };
  }, []);

  return (
    <div className="min-h-screen grid place-items-center bg-background px-6 text-center text-sm text-foreground/60">
      <div>
        <p>{error ? "the light could not cross the horizon" : "catching the light…"}</p>
        {error && <p className="mt-2 text-xs text-foreground/50">{error}</p>}
        {error && (
          <button
            type="button"
            onClick={() => window.location.replace("/")}
            className="mt-5 text-xs underline hover:text-foreground"
          >
            back to the horizon
          </button>
        )}
      </div>
    </div>
  );
}
